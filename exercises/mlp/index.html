def exercise1():
    x = np.array([0.5, -0.2])
    y = 1.0
    W1 = np.array([[0.3, -0.1],
                   [0.2, 0.4]])          # W^(1): linha i = pesos do neurônio oculto i
    b1 = np.array([0.1, -0.2])
    W2 = np.array([0.5, -0.3])           # W^(2): 1 neurônio de saída
    b2 = 0.2
    eta = 0.3

    # A — forward
    z1 = W1 @ x + b1
    h1 = np.tanh(z1)
    u2 = W2 @ h1 + b2
    y_hat = np.tanh(u2)
    L = (y - y_hat) ** 2

    # B — backward (regra da cadeia, tanh'(u) = 1 - tanh(u)^2)
    dL_dyhat = -2.0 * (y - y_hat)
    dL_du2 = dL_dyhat * (1.0 - y_hat ** 2)
    dL_dW2 = dL_du2 * h1
    dL_db2 = dL_du2
    dL_dh1 = dL_du2 * W2
    dL_dz1 = dL_dh1 * (1.0 - h1 ** 2)
    dL_dW1 = np.outer(dL_dz1, x)
    dL_db1 = dL_dz1

    # C — update
    W2_new = W2 - eta * dL_dW2
    b2_new = b2 - eta * dL_db2
    W1_new = W1 - eta * dL_dW1
    b1_new = b1 - eta * dL_db1

    # Segundo forward, para conferir a direção da perda
    h1_2 = np.tanh(W1_new @ x + b1_new)
    L_2 = (y - np.tanh(W2_new @ h1_2 + b2_new)) ** 2

    return dict(z1=z1, h1=h1, u2=u2, y_hat=y_hat, L=L,
                dL_dyhat=dL_dyhat, dL_du2=dL_du2, dL_dW2=dL_dW2, dL_db2=dL_db2,
                dL_dh1=dL_dh1, dL_dz1=dL_dz1, dL_dW1=dL_dW1, dL_db1=dL_db1,
                W2_new=W2_new, b2_new=b2_new, W1_new=W1_new, b1_new=b1_new,
                L_second_pass=L_2)


# =============================================================================
# O MLP — escrito do zero. A MESMA classe serve os Exercises 2, 3 e 4:
# só mudam os argumentos (layer_sizes, loss, lr), nunca o código.
# =============================================================================
def sigmoid(u):
    # Forma numericamente estável: evita exp de números grandes positivos.
    out = np.empty_like(u)
    pos = u >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-u[pos]))
    e = np.exp(u[~pos])
    out[~pos] = e / (1.0 + e)
    return out


def softmax(u):
    # Subtrair o máximo da linha não muda o resultado e evita overflow.
    e = np.exp(u - u.max(axis=1, keepdims=True))
    return e / e.sum(axis=1, keepdims=True)


def binary_cross_entropy(P, Y, eps=1e-12):
    P = np.clip(P, eps, 1 - eps)
    return float(-np.mean(Y * np.log(P) + (1 - Y) * np.log(1 - P)))


def categorical_cross_entropy(P, Y, eps=1e-12):
    return float(-np.mean(np.sum(Y * np.log(np.clip(P, eps, 1.0)), axis=1)))


# Cada perda vem com a ativação de saída que a "casa": sigmoid+BCE e softmax+CCE.
# Nos dois pares, dL/du da saída é (P - Y)/N — por isso o backward é um só.
LOSSES = {
    "bce": (sigmoid, binary_cross_entropy),
    "cce": (softmax, categorical_cross_entropy),
}


class MLP:
    """
    MLP totalmente conectado, tanh nas camadas ocultas.

    layer_sizes: [n_entradas, n_oculta_1, ..., n_oculta_k, n_saidas]
    loss:        "bce" (saída sigmoid, 1 unidade) ou "cce" (saída softmax, C unidades)

    Convenção de formas: X é (N, d); W[l] é (n_in, n_out), então a camada faz
    U = A_prev @ W + b  — é o W^(l) do enunciado transposto, aplicado a N amostras.
    """

    def __init__(self, layer_sizes, loss, lr, rng):
        self.sizes = list(layer_sizes)
        self.out_act, self.loss_fn = LOSSES[loss]
        self.lr = lr
        self.rng = rng
        # Inicialização Xavier/Glorot (normal), adequada a tanh.
        self.W, self.b = [], []
        for n_in, n_out in zip(self.sizes[:-1], self.sizes[1:]):
            std = np.sqrt(2.0 / (n_in + n_out))
            self.W.append(rng.normal(0.0, std, size=(n_in, n_out)))
            self.b.append(np.zeros(n_out))

    def n_params(self):
        return int(sum(W.size + b.size for W, b in zip(self.W, self.b)))

    # ---- forward ------------------------------------------------------------
    def forward(self, X):
        """Guarda as ativações de cada camada (A[0] = X) para o backward."""
        self.A = [X]
        for l, (W, b) in enumerate(zip(self.W, self.b)):
            U = self.A[-1] @ W + b
            last = l == len(self.W) - 1
            self.A.append(self.out_act(U) if last else np.tanh(U))
        return self.A[-1]

    # ---- backward -----------------------------------------------------------
    def backward(self, Y):
        """
        Gradientes de L (média sobre o batch) em relação a todos os W e b.
        delta = dL/dU da camada atual.
          saída:   delta = (P - Y) / N            (sigmoid+BCE ou softmax+CCE)
          camada:  dW = A_prev^T delta,  db = soma de delta nas amostras
          descer:  delta_prev = (delta W^T) ⊙ (1 - A_prev^2)   (derivada da tanh)
        """
        N = Y.shape[0]
        delta = (self.A[-1] - Y) / N
        self.dW = [None] * len(self.W)
        self.db = [None] * len(self.b)
        for l in range(len(self.W) - 1, -1, -1):
            A_prev = self.A[l]
            self.dW[l] = A_prev.T @ delta
            self.db[l] = delta.sum(axis=0)
            if l > 0:
                delta = (delta @ self.W[l].T) * (1.0 - A_prev ** 2)

    # ---- update -------------------------------------------------------------
    def step(self):
        """Gradiente descendente: θ ← θ - η ∂L/∂θ."""
        for l in range(len(self.W)):
            self.W[l] -= self.lr * self.dW[l]
            self.b[l] -= self.lr * self.db[l]

    # ---- treino / predição --------------------------------------------------
    def fit(self, X, Y, epochs, batch_size=32):
        """Mini-batch SGD; embaralha com o rng do relatório a cada época.
        Registra a perda no treino inteiro ao fim de cada época."""
        history = []
        N = X.shape[0]
        for _ in range(epochs):
            idx = self.rng.permutation(N)
            for start in range(0, N, batch_size):
                batch = idx[start:start + batch_size]
                self.forward(X[batch])
                self.backward(Y[batch])
                self.step()
            history.append(self.loss_fn(self.forward(X), Y))
        return np.array(history)

    def predict(self, X):
        P = self.forward(X)
        if P.shape[1] == 1:
            return (P[:, 0] >= 0.5).astype(int)
        return P.argmax(axis=1)


def one_hot(y, n_classes):
    return np.eye(n_classes)[y]

    def exercise1():
    x = np.array([0.5, -0.2])
    y = 1.0
    W1 = np.array([[0.3, -0.1],
                   [0.2, 0.4]])          # W^(1): linha i = pesos do neurônio oculto i
    b1 = np.array([0.1, -0.2])
    W2 = np.array([0.5, -0.3])           # W^(2): 1 neurônio de saída
    b2 = 0.2
    eta = 0.3

    # A — forward
    z1 = W1 @ x + b1
    h1 = np.tanh(z1)
    u2 = W2 @ h1 + b2
    y_hat = np.tanh(u2)
    L = (y - y_hat) ** 2

    # B — backward (regra da cadeia, tanh'(u) = 1 - tanh(u)^2)
    dL_dyhat = -2.0 * (y - y_hat)
    dL_du2 = dL_dyhat * (1.0 - y_hat ** 2)
    dL_dW2 = dL_du2 * h1
    dL_db2 = dL_du2
    dL_dh1 = dL_du2 * W2
    dL_dz1 = dL_dh1 * (1.0 - h1 ** 2)
    dL_dW1 = np.outer(dL_dz1, x)
    dL_db1 = dL_dz1

    # C — update
    W2_new = W2 - eta * dL_dW2
    b2_new = b2 - eta * dL_db2
    W1_new = W1 - eta * dL_dW1
    b1_new = b1 - eta * dL_db1

    # Segundo forward, para conferir a direção da perda
    h1_2 = np.tanh(W1_new @ x + b1_new)
    L_2 = (y - np.tanh(W2_new @ h1_2 + b2_new)) ** 2

    return dict(z1=z1, h1=h1, u2=u2, y_hat=y_hat, L=L,
                dL_dyhat=dL_dyhat, dL_du2=dL_du2, dL_dW2=dL_dW2, dL_db2=dL_db2,
                dL_dh1=dL_dh1, dL_dz1=dL_dz1, dL_dW1=dL_dW1, dL_db1=dL_db1,
                W2_new=W2_new, b2_new=b2_new, W1_new=W1_new, b1_new=b1_new,
                L_second_pass=L_2)

    def sigmoid(u):
    # Forma numericamente estável: evita exp de números grandes positivos.
    out = np.empty_like(u)
    pos = u >= 0
    out[pos] = 1.0 / (1.0 + np.exp(-u[pos]))
    e = np.exp(u[~pos])
    out[~pos] = e / (1.0 + e)
    return out


def softmax(u):
    # Subtrair o máximo da linha não muda o resultado e evita overflow.
    e = np.exp(u - u.max(axis=1, keepdims=True))
    return e / e.sum(axis=1, keepdims=True)


def binary_cross_entropy(P, Y, eps=1e-12):
    P = np.clip(P, eps, 1 - eps)
    return float(-np.mean(Y * np.log(P) + (1 - Y) * np.log(1 - P)))


def categorical_cross_entropy(P, Y, eps=1e-12):
    return float(-np.mean(np.sum(Y * np.log(np.clip(P, eps, 1.0)), axis=1)))


# Cada perda vem com a ativação de saída que a "casa": sigmoid+BCE e softmax+CCE.
# Nos dois pares, dL/du da saída é (P - Y)/N — por isso o backward é um só.
LOSSES = {
    "bce": (sigmoid, binary_cross_entropy),
    "cce": (softmax, categorical_cross_entropy),
}


class MLP:
    """
    MLP totalmente conectado, tanh nas camadas ocultas.

    layer_sizes: [n_entradas, n_oculta_1, ..., n_oculta_k, n_saidas]
    loss:        "bce" (saída sigmoid, 1 unidade) ou "cce" (saída softmax, C unidades)

    Convenção de formas: X é (N, d); W[l] é (n_in, n_out), então a camada faz
    U = A_prev @ W + b  — é o W^(l) do enunciado transposto, aplicado a N amostras.
    """

    def __init__(self, layer_sizes, loss, lr, rng):
        self.sizes = list(layer_sizes)
        self.out_act, self.loss_fn = LOSSES[loss]
        self.lr = lr
        self.rng = rng
        # Inicialização Xavier/Glorot (normal), adequada a tanh.
        self.W, self.b = [], []
        for n_in, n_out in zip(self.sizes[:-1], self.sizes[1:]):
            std = np.sqrt(2.0 / (n_in + n_out))
            self.W.append(rng.normal(0.0, std, size=(n_in, n_out)))
            self.b.append(np.zeros(n_out))

    def n_params(self):
        return int(sum(W.size + b.size for W, b in zip(self.W, self.b)))

    # ---- forward ------------------------------------------------------------
    def forward(self, X):
        """Guarda as ativações de cada camada (A[0] = X) para o backward."""
        self.A = [X]
        for l, (W, b) in enumerate(zip(self.W, self.b)):
            U = self.A[-1] @ W + b
            last = l == len(self.W) - 1
            self.A.append(self.out_act(U) if last else np.tanh(U))
        return self.A[-1]

    # ---- backward -----------------------------------------------------------
    def backward(self, Y):
        """
        Gradientes de L (média sobre o batch) em relação a todos os W e b.
        delta = dL/dU da camada atual.
          saída:   delta = (P - Y) / N            (sigmoid+BCE ou softmax+CCE)
          camada:  dW = A_prev^T delta,  db = soma de delta nas amostras
          descer:  delta_prev = (delta W^T) ⊙ (1 - A_prev^2)   (derivada da tanh)
        """
        N = Y.shape[0]
        delta = (self.A[-1] - Y) / N
        self.dW = [None] * len(self.W)
        self.db = [None] * len(self.b)
        for l in range(len(self.W) - 1, -1, -1):
            A_prev = self.A[l]
            self.dW[l] = A_prev.T @ delta
            self.db[l] = delta.sum(axis=0)
            if l > 0:
                delta = (delta @ self.W[l].T) * (1.0 - A_prev ** 2)

    # ---- update -------------------------------------------------------------
    def step(self):
        """Gradiente descendente: θ ← θ - η ∂L/∂θ."""
        for l in range(len(self.W)):
            self.W[l] -= self.lr * self.dW[l]
            self.b[l] -= self.lr * self.db[l]

    # ---- treino / predição --------------------------------------------------
    def fit(self, X, Y, epochs, batch_size=32):
        """Mini-batch SGD; embaralha com o rng do relatório a cada época.
        Registra a perda no treino inteiro ao fim de cada época."""
        history = []
        N = X.shape[0]
        for _ in range(epochs):
            idx = self.rng.permutation(N)
            for start in range(0, N, batch_size):
                batch = idx[start:start + batch_size]
                self.forward(X[batch])
                self.backward(Y[batch])
                self.step()
            history.append(self.loss_fn(self.forward(X), Y))
        return np.array(history)

    def predict(self, X):
        P = self.forward(X)
        if P.shape[1] == 1:
            return (P[:, 0] >= 0.5).astype(int)
        return P.argmax(axis=1)

    D — Analysis

    A Figura 3 mostra a resposta: a região que a rede atribui à Classe 0 é uma faixa diagonal que passa pela origem, e a Classe 1 recebe as duas regiões de fora dessa faixa — uma no canto superior direito (o blob em \((3,3)\)) e outra no inferior esquerdo (o blob em \((-3,-3)\)). A região laranja é, portanto, desconexa.

    Uma única reta divide o plano em dois semiplanos, e cada semiplano é convexo e conexo. Não há como pôr \((3,3)\) e \((-3,-3)\) do mesmo lado de uma reta deixando \((0,0)\) — o ponto médio dos dois — do outro lado: se os dois extremos estão num semiplano, o segmento entre eles também está, e \((0,0)\) está nesse segmento. Qualquer reta que o modelo linear escolha deixa, então, pelo menos um dos blobs da Classe 1 do lado "errado".

    Por que abaixo de 50%? Olhando o que a regressão logística aprendeu, ela essencialmente desistiu: \(w = [0.0296,\ 0.0015]\), \(b = -0.0031\), e as probabilidades no teste ficam todas entre 0.457 e 0.544. A fronteira \(p=0.5\) é praticamente a reta vertical \(x_1 \approx 0\). O motivo é a simetria: qualquer inclinação que ganhe um blob da Classe 1 perde o outro na mesma medida, então a log-loss quase não muda e o ótimo é uma rede quase "plana". O que essa reta faz no teste:

    Blob (teste) Classe Pontos Acertos da reta
    \((0,0)\) 0 100 53 (cortado ao meio)
    \((3,3)\) 1 42 42 (todo do lado "1")
    \((-3,-3)\) 1 58 0 (todo do lado "0")
    Total 200 95 → 47.5%

    Ou seja, a reta acerta o blob central pela metade, acerta um blob externo inteiro e erra o outro inteiro. O resultado ficaria em ~50% se os dois blobs externos tivessem o mesmo tamanho no teste; como o split estratifica por y (e não por blob), o blob errado ficou com 58 pontos e o certo com 42, e isso empurra a acurácia para baixo de 50%. É a geometria da Figura 1 em ação: com uma reta, um dos dois blobs da Classe 1 fica obrigatoriamente do lado errado.

    A camada oculta resolve isso porque cada um dos 8 neurônios tanh é ele mesmo uma "reta suave"; a camada de saída combina essas retas. Duas delas, paralelas à diagonal \(x_1+x_2=\text{const}\) e com sinais opostos, já bastam para cercar a faixa central — é exatamente o formato que se vê na Figura 3 (as duas bordas da faixa são aproximadamente paralelas). Os ~8% de erro que restam estão onde as nuvens de fato se sobrepõem (cluster_std=1.2, a ~4.2 unidades de distância): pontos laranja dentro da faixa e azuis fora dela, que nenhum classificador separaria sem decorar ruído.

    B — Adapt the network

    As únicas mudanças em relação ao Exercise 2 são argumentos: layer_sizes termina em 3 unidades e loss="cce", que liga a saída softmax e a categorical cross-entropy \(L = -\frac1N\sum_n\sum_k y_{nk}\log p_{nk}\). Forward, backward e step são exatamente as mesmas linhas do Ex. 2 (é a mesma classe MLP).

    Derivação de \(\partial L/\partial u\) para softmax + CCE. Para uma amostra, com \(p_k = e^{u_k}/\sum_j e^{u_j}\) e \(L = -\sum_k y_k\log p_k\) (\(y\) one-hot).

    Jacobiano da softmax. Derivando o quociente:

    \[ \frac{\partial p_k}{\partial u_j} = \begin{cases} p_k(1-p_k) & k=j\\ -p_kp_j & k\neq j\end{cases} \;=\; p_k(\delta_{kj} - p_j) \]

    Regra da cadeia. Como \(\partial L/\partial p_k = -y_k/p_k\):

    \[ \frac{\partial L}{\partial u_j} = \sum_k \frac{\partial L}{\partial p_k}\frac{\partial p_k}{\partial u_j} = \sum_k -\frac{y_k}{p_k}\,p_k(\delta_{kj}-p_j) = -\sum_k y_k\delta_{kj} + p_j\sum_k y_k = -y_j + p_j\cdot 1 \]
    \[ \boxed{\;\frac{\partial L}{\partial u} = p - y\;} \qquad\text{(e } (P-Y)/N \text{ com a média sobre o batch)} \]

    Por que é tão simples. Dois motivos se somam. (i) O \(\log\) da CCE desfaz a exponencial da softmax: o \(1/p_k\) de \(\partial L/\partial p_k\) cancela exatamente o fator \(p_k\) do jacobiano, então não sobra nenhum termo de saturação — ao contrário do Ex. 1, onde o \((1-\hat y^2)\) da tanh multiplica o gradiente e o encolhe quando a saída satura. (ii) \(\sum_k y_k = 1\), o que junta os termos cruzados num só \(p_j\). O resultado tem leitura direta: o sinal de erro de cada unidade é "probabilidade prevista menos alvo". O mesmo vale para sigmoid + BCE (\(\sigma'(u) = \sigma(1-\sigma)\) cancela com o denominador da BCE e dá \(p-y\)) — e é por isso que uma única linha delta = (self.A[-1] - Y) / N serve aos Exercises 2, 3 e 4 sem mudança, o que caracteriza o reuso literal pedido no item extra.

    O que medi. Só que o teste não acompanha: a distância treino–teste passa de ~4.2 p.p. para ~6.2 p.p. Os 272 parâmetros extras estão sendo gastos para ajustar melhor o treino — inclusive pontos nas regiões de sobreposição e os rótulos trocados pelo flip_y — e não para aprender estrutura que generalize.

    Por que a profundidade não paga aqui. O dataset tem só 6 clusters gaussianos em vértices de um hipercubo 4D, e cada classe é a união de dois deles. A fronteira necessária é "recortar alguns blobs", algo que uma camada oculta com 16 neurônios já representa bem (cada neurônio é um hiperplano; a saída combina as regiões). O que limita a acurácia não é capacidade, mas a sobreposição entre clusters vizinhos (as arestas C0–C1, C1–C2, C2–C4 do Ex. 3.D) e o ruído de rótulo: o erro restante é praticamente irredutível, e nenhuma arquitetura o remove. Em termos de representação, a segunda camada permite compor features — combinar hiperplanos em regiões e depois combinar regiões —, mas não há hierarquia a compor quando a estrutura é "6 blobs". Em termos de otimização, ela custa: 3× mais parâmetros com os mesmos 1200 exemplos, uma superfície de perda mais irregular (a curva laranja da Figura 6 é visivelmente mais ruidosa) e mais sensibilidade à inicialização.

    Que dado faria a profundidade valer a pena. Dados com estrutura composicional e hierárquica, em que a fronteira é uma função de funções: imagens (bordas → texturas → partes → objetos), áudio, texto; ou, em tabelas, problemas do tipo paridade/XOR de muitas variáveis, espirais entrelaçadas, ou tabuleiros de xadrez com muitas células — fronteiras que uma rede rasa só aproxima com um número exponencial de neurônios, mas que uma rede profunda monta reutilizando features intermediárias. Também ajudaria ter muito mais exemplos, para que a capacidade extra fosse usada em estrutura real e não em sobreajuste.
